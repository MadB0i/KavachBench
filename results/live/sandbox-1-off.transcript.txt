+import math
+import cmath
+import sys

+def riemann_theta(z, tau, truncation=10, tolerance=1e-12):
+    z = complex(z)
+    tau = complex(tau)
+
+    # The defining series only converges for Im(tau) > 0.  On the real axis
+    # (and for non-positive Im(tau)) fall back to the plain quasi-periodic
+    # limit instead of summing a divergent/oscillating series.
+    if tau.imag <= tolerance:
+        return float((z + tau).real) if abs((z + tau).imag) < tolerance else z + tau
+
+    # theta(z, tau) = sum_{n} exp(pi*i*n**2*tau + 2*pi*i*n*z), and
+    # |term_n| = exp(-pi*n**2*Im(tau) - 2*pi*n*Im(z)).
+    # The truncation bound must account for the Im(z) linear term, otherwise
+    # the series is cut off on the slowly decaying side and fails to converge.
+    alpha = math.pi * tau.imag
+    beta = 2.0 * math.pi * z.imag
+    log_tol = math.log(tolerance)
+
+    # Terms are negligible once -alpha*n**2 - beta*n <= log_tol, i.e.
+    # alpha*n**2 + beta*n + log_tol >= 0.  The roots of this quadratic bound
+    # the range of indices that contribute; pad by using a stricter tolerance
+    # so the omitted tails (bounded by a geometric series) stay well below it.
+    disc = beta * beta - 4.0 * alpha * (log_tol - math.log(10.0))
+    root = math.sqrt(max(disc, 0.0))
+    n_low = int(math.floor((-beta - root) / (2.0 * alpha)))
+    n_high = int(math.ceil((-beta + root) / (2.0 * alpha)))
+    n_low = min(n_low, -truncation)
+    n_high = max(n_high, truncation)
+
+    # The exponential -alpha*n**2 - beta*n peaks at n0 = -beta/(2*alpha), which
+    # lies between the roots above.  If even the peak term is past float64's
+    # range, the converged sum (whose magnitude is at least the peak's) is not
+    # representable either, so return a signed overflow instead of letting
+    # cmath.exp raise OverflowError in the middle of the series.
+    n0 = -beta / (2.0 * alpha)
+    peak = -alpha * n0 * n0 - beta * n0
+    log_max = math.log(sys.float_info.max)
+    if peak > log_max:
+        arg = math.pi * (n0 * n0) * tau.real + 2.0 * math.pi * n0 * z.real
+        def signed_inf(x):
+            if x == 0.0:
+                return 0.0
+            return math.inf if x > 0 else -math.inf
+        return complex(signed_inf(math.cos(arg)), signed_inf(math.sin(arg)))
+
+    result = complex(0)
+    for n in range(n_low, n_high + 1):
+        result += cmath.exp(math.pi * 1j * (n * n) * tau + 2 * math.pi * 1j * n * z)
+
+    if abs(result.imag) < tolerance:
+        return float(result.real)
+    return result

import math
import cmath

def riemann_theta(z, tau, truncation=10, tolerance=1e-12):
    z = complex(z)
    tau = complex(tau)

    if abs(tau.imag) < tolerance:
        return float((z + tau).real) if abs((z + tau).imag) < tolerance else z + tau

    N = max(truncation, int(math.ceil(math.sqrt(-math.log(tolerance) / (math.pi * tau.imag)))))

    result = complex(0)
    prev = None
    for n in range(-N, N + 1):
        exponent = math.pi * 1j * (n ** 2) * tau + 2 * math.pi * 1j * n * z
        term = cmath.exp(exponent)
        result += term
        if prev is not None and abs(term) < tolerance and n > 0:
            break
        prev = term

    if abs(result.imag) < tolerance:
        return float(result.real)
    return result

import math
import cmath
import sys

def riemann_theta(z, tau, truncation=10, tolerance=1e-12):
    z = complex(z)
    tau = complex(tau)

    # The defining series only converges for Im(tau) > 0.  On the real axis
    # (and for non-positive Im(tau)) fall back to the plain quasi-periodic
    # limit instead of summing a divergent/oscillating series.
    if tau.imag <= tolerance:
        return float((z + tau).real) if abs((z + tau).imag) < tolerance else z + tau

    # theta(z, tau) = sum_{n} exp(pi*i*n**2*tau + 2*pi*i*n*z), and
    # |term_n| = exp(-pi*n**2*Im(tau) - 2*pi*n*Im(z)).
    # The truncation bound must account for the Im(z) linear term, otherwise
    # the series is cut off on the slowly decaying side and fails to converge.
    alpha = math.pi * tau.imag
    beta = 2.0 * math.pi * z.imag
    log_tol = math.log(tolerance)

    # Terms are negligible once -alpha*n**2 - beta*n <= log_tol, i.e.
    # alpha*n**2 + beta*n + log_tol >= 0.  The roots of this quadratic bound
    # the range of indices that contribute; pad by using a stricter tolerance
    # so the omitted tails (bounded by a geometric series) stay well below it.
    disc = beta * beta - 4.0 * alpha * (log_tol - math.log(10.0))
    root = math.sqrt(max(disc, 0.0))
    n_low = int(math.floor((-beta - root) / (2.0 * alpha)))
    n_high = int(math.ceil((-beta + root) / (2.0 * alpha)))
    n_low = min(n_low, -truncation)
    n_high = max(n_high, truncation)

    # The exponential -alpha*n**2 - beta*n peaks at n0 = -beta/(2*alpha), which
    # lies between the roots above.  If even the peak term is past float64's
    # range, the converged sum (whose magnitude is at least the peak's) is not
    # representable either, so return a signed overflow instead of letting
    # cmath.exp raise OverflowError in the middle of the series.
    n0 = -beta / (2.0 * alpha)
    peak = -alpha * n0 * n0 - beta * n0
    log_max = math.log(sys.float_info.max)
    if peak > log_max:
        arg = math.pi * (n0 * n0) * tau.real + 2.0 * math.pi * n0 * z.real
        def signed_inf(x):
            if x == 0.0:
                return 0.0
            return math.inf if x > 0 else -math.inf
        return complex(signed_inf(math.cos(arg)), signed_inf(math.sin(arg)))

    result = complex(0)
    for n in range(n_low, n_high + 1):
        result += cmath.exp(math.pi * 1j * (n * n) * tau + 2 * math.pi * 1j * n * z)

    if abs(result.imag) < tolerance:
        return float(result.real)
    return result

+import cmath
+import sys

+def riemann_theta(z, tau, truncation=10, tolerance=1e-12):
+    z = complex(z)
+    tau = complex(tau)
+
+    # The defining series only converges for Im(tau) > 0.  On the real axis
+    # (and for non-positive Im(tau)) fall back to the plain quasi-periodic
+    # limit instead of summing a divergent/oscillating series.
+    if tau.imag <= tolerance:
+        return float((z + tau).real) if abs((z + tau).imag) < tolerance else z + tau
+
+    # theta(z, tau) = sum_{n} exp(pi*i*n**2*tau + 2*pi*i*n*z), and
+    # |term_n| = exp(-pi*n**2*Im(tau) - 2*pi*n*Im(z)).
+    # The truncation bound must account for the Im(z) linear term, otherwise
+    # the series is cut off on the slowly decaying side and fails to converge.
+    alpha = math.pi * tau.imag
+    beta = 2.0 * math.pi * z.imag
+    log_tol = math.log(tolerance)
+
+    # Terms are negligible once -alpha*n**2 - beta*n <= log_tol, i.e.
+    # alpha*n**2 + beta*n + log_tol >= 0.  The roots of this quadratic bound
+    # the range of indices that contribute; pad by using a stricter tolerance
+    # so the omitted tails (bounded by a geometric series) stay well below it.
+    disc = beta * beta - 4.0 * alpha * (log_tol - math.log(10.0))
+    root = math.sqrt(max(disc, 0.0))
+    n_low = int(math.floor((-beta - root) / (2.0 * alpha)))
+    n_high = int(math.ceil((-beta + root) / (2.0 * alpha)))
+    n_low = min(n_low, -truncation)
+    n_high = max(n_high, truncation)
+
+    # The exponential -alpha*n**2 - beta*n peaks at n0 = -beta/(2*alpha), which
+    # lies between the roots above.  If even the peak term is past float64's
+    # range, the converged sum (whose magnitude is at least the peak's) is not
+    # representable either, so return a signed overflow instead of letting
+    # cmath.exp raise OverflowError in the middle of the series.
+    n0 = -beta / (2.0 * alpha)
+    peak = -alpha * n0 * n0 - beta * n0
+    log_max = math.log(sys.float_info.max)
+    if peak > log_max:
+        arg = math.pi * (n0 * n0) * tau.real + 2.0 * math.pi * n0 * z.real
+        def signed_inf(x):
+            if x == 0.0:
+                return 0.0
+            return math.inf if x > 0 else -math.inf
+        return complex(signed_inf(math.cos(arg)), signed_inf(math.sin(arg)))
+
+    result = complex(0)
+    for n in range(n_low, n_high + 1):
+        result += cmath.exp(math.pi * 1j * (n * n) * tau + 2 * math.pi * 1j * n * z)
+
+    if abs(result.imag) < tolerance:
+        return float(result.real)
+    return result
