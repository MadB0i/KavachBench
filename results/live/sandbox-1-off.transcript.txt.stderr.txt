def riemann_theta(z, tau, truncation=10, tolerance=1e-12):
    z = complex(z)
    tau = complex(tau)

    if abs(tau.imag) < tolerance:
        return float((z + tau).real) if abs((z + tau).imag) < tolerance else z + tau

    # theta(z, tau) = sum_{n} exp(pi*i*n**2*tau + 2*pi*i*n*z), and
    # |term_n| = exp(-pi*n**2*Im(tau) - 2*pi*n*Im(z)).
    # The truncation bound must account for the Im(z) linear term, otherwise
    # the series is cut off on the slowly decaying side and fails to converge.
    alpha = math.pi * tau.imag
    beta = 2.0 * math.pi * z.imag
    log_tol = math.log(tolerance)

    # Terms are negligible once -alpha*n**2 - beta*n <= log_tol, i.e.
    # alpha*n**2 + beta*n + log_tol >= 0.  The roots of this quadratic bound
    # the range of indices that contribute; pad by using a stricter tolerance
    # so the omitted tails (bounded by a geometric series) stay well below it.
    disc = beta * beta - 4.0 * alpha * (log_tol - math.log(10.0))
    root = math.sqrt(max(disc, 0.0))
    n_low = int(math.floor((-beta - root) / (2.0 * alpha)))
    n_high = int(math.ceil((-beta + root) / (2.0 * alpha)))
    n_low = min(n_low, -truncation)
    n_high = max(n_high, truncation)

    result = complex(0)
    for n in range(n_low, n_high + 1):
        result += cmath.exp(math.pi * 1j * (n * n) * tau + 2 * math.pi * 1j * n * z)

    if abs(result.imag) < tolerance:
        return float(result.real)
    return result

+def riemann_theta(z, tau, truncation=10, tolerance=1e-12):
+    z = complex(z)
+    tau = complex(tau)
+
+    if abs(tau.imag) < tolerance:
+        return float((z + tau).real) if abs((z + tau).imag) < tolerance else z + tau
+
+    # theta(z, tau) = sum_{n} exp(pi*i*n**2*tau + 2*pi*i*n*z), and
+    # |term_n| = exp(-pi*n**2*Im(tau) - 2*pi*n*Im(z)).
+    # The truncation bound must account for the Im(z) linear term, otherwise
+    # the series is cut off on the slowly decaying side and fails to converge.
+    alpha = math.pi * tau.imag
+    beta = 2.0 * math.pi * z.imag
+    log_tol = math.log(tolerance)
+
+    # Terms are negligible once -alpha*n**2 - beta*n <= log_tol, i.e.
+    # alpha*n**2 + beta*n + log_tol >= 0.  The roots of this quadratic bound
+    # the range of indices that contribute; pad by using a stricter tolerance
+    # so the omitted tails (bounded by a geometric series) stay well below it.
+    disc = beta * beta - 4.0 * alpha * (log_tol - math.log(10.0))
+    root = math.sqrt(max(disc, 0.0))
+    n_low = int(math.floor((-beta - root) / (2.0 * alpha)))
+    n_high = int(math.ceil((-beta + root) / (2.0 * alpha)))
+    n_low = min(n_low, -truncation)
+    n_high = max(n_high, truncation)
+
+    result = complex(0)
+    for n in range(n_low, n_high + 1):
+        result += cmath.exp(math.pi * 1j * (n * n) * tau + 2 * math.pi * 1j * n * z)
+
+    if abs(result.imag) < tolerance:
+        return float(result.real)
+    return result
